cmake_minimum_required(VERSION 3.22)
project(intercom C)
include(FetchContent)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MACOSX_BUNDLE OFF)
set(LIBRE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(USE_OPENSSL OFF CACHE BOOL "" FORCE)
set(USE_TLS1_3_PHA OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_OpenSSL ON)
set(CMAKE_DISABLE_FIND_PACKAGE_ZLIB ON)
set(STATIC ON CACHE BOOL "" FORCE)
if(ANDROID)
  set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(mbedtls GIT_REPOSITORY https://github.com/Mbed-TLS/mbedtls.git GIT_TAG c3db4124ce35e6a698019bd62a281f16ddb677a6)
  FetchContent_MakeAvailable(mbedtls)
  set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
  set(MBEDTLS_INCLUDE_DIR "${mbedtls_SOURCE_DIR}/include" CACHE PATH "" FORCE)
  set(MBEDTLS_LIBRARY mbedcrypto CACHE STRING "" FORCE)
  include_directories("${mbedtls_SOURCE_DIR}/include")
  set(MODULES "g711;aaudio;auconv;auresamp" CACHE STRING "" FORCE)
else()
  set(MODULES "g711;audiounit;auconv;auresamp" CACHE STRING "" FORCE)
endif()
FetchContent_Declare(libre GIT_REPOSITORY https://github.com/baresip/re.git GIT_TAG 65555dad439de9d6a81b3ee6092b9c096585249b)
FetchContent_MakeAvailable(libre)
list(APPEND CMAKE_MODULE_PATH "${libre_SOURCE_DIR}/cmake")
set(RE_INCLUDE_DIR "${libre_SOURCE_DIR}/include" CACHE PATH "" FORCE)
set(RE_LIBRARY re CACHE STRING "" FORCE)
set(re_DIR "${libre_SOURCE_DIR}/cmake" CACHE PATH "" FORCE)
FetchContent_Declare(baresip GIT_REPOSITORY https://github.com/baresip/baresip.git GIT_TAG 9cfab81550acc4bd38d6580d79b45cc6eea65cd6)
FetchContent_MakeAvailable(baresip)
if(ANDROID)
  add_library(intercom SHARED intercom.c android.c)
  target_link_libraries(intercom PRIVATE baresip re mbedcrypto aaudio log)
else()
  add_library(intercom STATIC intercom.c)
  target_link_libraries(intercom PRIVATE baresip re)
endif()
target_include_directories(intercom PRIVATE "${libre_SOURCE_DIR}/include" "${baresip_SOURCE_DIR}/include")
target_compile_options(intercom PRIVATE -Wall -Wextra -Werror)
